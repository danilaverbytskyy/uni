
Процедура Печать(ТабДок, Ссылка) Экспорт
	//{{_КОНСТРУКТОР_ПЕЧАТИ(Печать)
	Макет = Документы.ПриходнаяНакладная.ПолучитьМакет("Печать");
	Запрос = Новый Запрос;
	Запрос.Текст =
	"ВЫБРАТЬ
	|	ПриходнаяНакладная.Дата,
	|	ПриходнаяНакладная.Номер,
	|	ПриходнаяНакладная.Поставщик,
	|	ПриходнаяНакладная.Склад,
	|	ПриходнаяНакладная.Телефон_Поставщика,
	|	ПриходнаяНакладная.Приход.(
	|		НомерСтроки,
	|		Номенкулатура,
	|		Количество,
	|		Единицы_Измерения,
	|		Цена,
	|		Сумма
	|	)
	|ИЗ
	|	Документ.ПриходнаяНакладная КАК ПриходнаяНакладная
	|ГДЕ
	|	ПриходнаяНакладная.Ссылка В (&Ссылка)";
	Запрос.Параметры.Вставить("Ссылка", Ссылка);
	Выборка = Запрос.Выполнить().Выбрать();

	ОбластьЗаголовок = Макет.ПолучитьОбласть("Заголовок");
	Шапка = Макет.ПолучитьОбласть("Шапка");
	ОбластьПриходШапка = Макет.ПолучитьОбласть("ПриходШапка");
	ОбластьПриход = Макет.ПолучитьОбласть("Приход");
	ОбластьИтог = Макет.ПолучитьОбласть("Всего"); // я добавлял
	ТабДок.Очистить();

	ВставлятьРазделительСтраниц = Ложь;
	Пока Выборка.Следующий() Цикл
		Если ВставлятьРазделительСтраниц Тогда
			ТабДок.ВывестиГоризонтальныйРазделительСтраниц();
		КонецЕсли;

		ТабДок.Вывести(ОбластьЗаголовок);

		Шапка.Параметры.Заполнить(Выборка);
		ТабДок.Вывести(Шапка, Выборка.Уровень());

		ТабДок.Вывести(ОбластьПриходШапка);
		ВыборкаПриход = Выборка.Приход.Выбрать();
		СуммаИтог = 0; // я добавил
		Пока ВыборкаПриход.Следующий() Цикл
			ОбластьПриход.Параметры.Заполнить(ВыборкаПриход);
			ТабДок.Вывести(ОбластьПриход, ВыборкаПриход.Уровень());
			СуммаИтог = СуммаИтог +	ВыборкаПриход.Сумма; // я добавлял
		КонецЦикла;
        ОбластьИтог.Параметры.ВсегоПоДокументу = СуммаИтог; // я добавлял
		ТабДок.Вывести(ОбластьИтог); // я добавлял
		ВставлятьРазделительСтраниц = Истина;
	КонецЦикла;
	//}}
